🛠️ Si table manquante, exécute dans Supabase SQL:

create table if not exists public.secteur_sync (
 id text primary key,
 nom text, referent text, email text, region text, depts text,
 nb_aci int default 0, data jsonb default '{}'::jsonb,
 updated_at timestamptz default now()
);
alter table public.secteur_sync enable row level security;
drop policy if exists "allow all" on public.secteur_sync;
create policy "allow all" on public.secteur_sync for all using (true) with check (true);
alter publication supabase_realtime add table public.secteur_sync;

-- Optionnel: vue aci_secteurs = secteur_sync
create or replace view public.aci_secteurs as select * from public.secteur_sync;
